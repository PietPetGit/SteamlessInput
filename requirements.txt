# Runtime dependencies for both platforms.
#
# A single file serves the Windows and Linux CI jobs: platform-specific deps are
# guarded with PEP 508 environment markers, so `pip install -r requirements.txt`
# pulls the right set on each runner (see .github/workflows/build.yml).

# ---- shared ----
# (No PySDL2  rendering/input go through the vendored hand-rolled sdl3w ctypes
#  binding: Windows ships SDL3 DLLs in windows/sdl3w/dll; Linux loads the system
#  libSDL3 / libSDL3_ttf.)
Pillow==12.2.0
pynput==1.8.2
hidapi==0.15.0
PyYAML==6.0.3
pystray==0.19.5
# psutil is imported LAZILY (inside functions) at ~12 sites  Steam-running
# detection, auto-gamepad process scanning, launcher-child tracking. Every one
# of those sites catches ImportError and degrades silently, so a build made
# WITHOUT psutil installed doesn't fail  it just ships with "pause while
# Steam is running" (default ON) and Auto Gamepad Mode dead. That was exactly
# the state of every CI release until this line was added: the dev machine had
# psutil installed globally, CI didn't, and PyInstaller can't bundle what
# isn't there.
psutil==7.2.2

# ---- Windows only ----
# vgamepad is vendored under windows/vgamepad because upstream publishes only
# an sdist whose setup.py tries to launch the ViGEmBus MSI during CI installs.
#
# WinRT projections for the tutorial's silent demo track (windows/media_demo.py:
# a System Media Transport Controls session, so the Media Controls slide's
# Next/Play-Pause keys land on US and not on the user's own music). Imported
# lazily and wrapped, exactly like psutil above  a build made without these
# still runs, it just loses the now-playing card on that one slide. Which is
# precisely the failure psutil's note describes, so they are pinned here rather
# than left to the dev machine happening to have them.
winrt-Windows.Foundation ; sys_platform == "win32"
winrt-Windows.Media ; sys_platform == "win32"
winrt-Windows.Media.Core ; sys_platform == "win32"
winrt-Windows.Media.Playback ; sys_platform == "win32"
winrt-Windows.Storage ; sys_platform == "win32"
winrt-Windows.Storage.Streams ; sys_platform == "win32"

# ---- Linux only ----
PyGObject<3.52 ; sys_platform == "linux" # gi / AppIndicator tray backend (linux/tray_linux.py)

# ---- build tool (used by CI to produce the release binaries) ----
pyinstaller==6.20.0
